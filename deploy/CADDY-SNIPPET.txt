# Bloque para añadir a /etc/caddy/Caddyfile en el server de despliegue
# (no se commitea al repo porque /etc/caddy/Caddyfile es config del sistema)

dgaraje.andresmorales.com.co {
	encode gzip
	tls /etc/caddy/certs/dgaraje.andresmorales.com.co.fullchain.crt /etc/caddy/certs/dgaraje.andresmorales.com.co.key

	handle /api/webhooks/* {
		reverse_proxy 127.0.0.1:3035 {
			header_up X-Real-IP {remote_host}
			header_up X-Forwarded-For {remote_host}
			transport http {
				dial_timeout 10s
				response_header_timeout 60s
				read_timeout 60s
			}
		}
	}

	reverse_proxy 127.0.0.1:3035 {
		header_up X-Real-IP {remote_host}
		header_up X-Forwarded-For {remote_host}
		transport http {
			dial_timeout 10s
			response_header_timeout 60s
			read_timeout 60s
		}
	}

	header {
		Strict-Transport-Security "max-age=31536000; includeSubDomains; preload"
		X-Frame-Options "DENY"
		X-Content-Type-Options "nosniff"
		Referrer-Policy "strict-origin-when-cross-origin"
		Permissions-Policy "geolocation=(), camera=(), microphone=(), payment=(), usb=(), magnetometer=(), gyroscope=(), accelerometer=(), interest-cohort=()"
		-X-Powered-By
	}
}
